Calendar

Paste your calendar's secret iCal address; the sleep screen keeps it updated on the reader.

  1. Open calendar.google.com on a computer.
  2. Hover your calendar in the left list → ⋮ → Settings and sharing.
  3. Scroll to Integrate calendar and copy the Secret address in iCal format.
Organization account, or network blocks Google scripts?
  1. Create a GitHub token: open github.com/settings/tokens/new?scopes=gist, click Generate token, copy it.
  2. At script.google.com, replace your project's code with the script below and paste your token into the first line.
  3. Pick setup in the function dropdown, press Run, approve the popups, then copy the link printed in the log.
  4. Add a trigger (clock icon): function pushCalendar, time-driven, every 15 minutes.
  5. Paste the copied link into the box above and Save.
const GITHUB_TOKEN = "PASTE_YOUR_TOKEN_HERE";

function setup() {
  const r = UrlFetchApp.fetch("https://api.github.com/gists", {
    method: "post",
    headers: { Authorization: "Bearer " + GITHUB_TOKEN },
    contentType: "application/json",
    payload: JSON.stringify({ description: "CrossPoint calendar", public: false,
      files: { "calendar.ics": { content: buildIcs() } } })
  });
  const d = JSON.parse(r.getContentText());
  PropertiesService.getScriptProperties().setProperty("GIST_ID", d.id);
  Logger.log("PASTE THIS INTO THE READER: https://gist.githubusercontent.com/" + d.owner.login + "/" + d.id + "/raw/calendar.ics");
}

function pushCalendar() {
  const id = PropertiesService.getScriptProperties().getProperty("GIST_ID");
  if (!id) return;
  UrlFetchApp.fetch("https://api.github.com/gists/" + id, {
    method: "patch",
    headers: { Authorization: "Bearer " + GITHUB_TOKEN },
    contentType: "application/json",
    payload: JSON.stringify({ files: { "calendar.ics": { content: buildIcs() } } })
  });
}

function buildIcs() {
  const cal = CalendarApp.getDefaultCalendar();
  const now = new Date();
  const end = new Date(now.getTime() + 14 * 24 * 60 * 60 * 1000);
  const events = cal.getEvents(now, end);
  const fmt = d => Utilities.formatDate(d, "UTC", "yyyyMMdd'T'HHmmss'Z'");
  const fmtDay = d => Utilities.formatDate(d, cal.getTimeZone(), "yyyyMMdd");
  let out = "BEGIN:VCALENDAR\r\nVERSION:2.0\r\nPRODID:-//CrossPoint bridge//EN\r\n";
  for (const e of events) {
    out += "BEGIN:VEVENT\r\n";
    if (e.isAllDayEvent()) {
      out += "DTSTART;VALUE=DATE:" + fmtDay(e.getAllDayStartDate()) + "\r\n";
      out += "DTEND;VALUE=DATE:" + fmtDay(e.getAllDayEndDate()) + "\r\n";
    } else {
      out += "DTSTART:" + fmt(e.getStartTime()) + "\r\n";
      out += "DTEND:" + fmt(e.getEndTime()) + "\r\n";
    }
    out += "SUMMARY:" + e.getTitle().replace(/([\\;,])/g, "\\$1").replace(/\n/g, "\\n") + "\r\n";
    out += "UID:" + e.getId() + "\r\n";
    out += "END:VEVENT\r\n";
  }
  out += "END:VCALENDAR\r\n";
  return out;
}
